```typescript
"use client";

import { useState } from "react";
import { JsonAnalysisPanel } from "../components/json-analysis/json-analysis-panel";
import { AudioModePanel } from "../components/audio-mode/audio-mode-panel";

type AppMode = "json" | "audio";

export default function Home() {
  const [mode, setMode] = useState<AppMode>("json");

  return (
    <div className="min-h-screen p-4">
      {/* Mode tabs */}
      <div className="flex gap-2 mb-6 border-b border-gray-200">
        <button
          onClick={() => setMode("json")}
          className={`px-4 py-2 font-medium transition-colors ${
            mode === "json"
              ? "border-b-2 border-blue-600 text-blue-600"
              : "text-gray-500 hover:text-gray-700"
          }`}
        >
          JSON Mode
        </button>
        <button
          onClick={() => setMode("audio")}
          className={`px-4 py-2 font-medium transition-colors ${
            mode === "audio"
              ? "border-b-2 border-blue-600 text-blue-600"
              : "text-gray-500 hover:text-gray-700"
          }`}
        >
          Audio Mode
        </button>
      </div>

      {/* Mode content */}
      {mode === "json" && <JsonAnalysisPanel />}
      {mode === "audio" && <AudioModePanel />}
    </div>
  );
}
```

Key points:
- Page becomes a client component (needs useState for mode switching)
- D-23: Audio mode is a separate top-level tab/view, not nested inside JSON analysis
- Both panels render conditionally (not just hidden with CSS — unmount when switching to keep state fresh per D-07)
- Styling: simple tab UI with active indicator underline
