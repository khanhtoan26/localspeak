```typescript
/**
 * Linear interpolation resampling from source rate to target rate.
 * Sufficient quality for speech at 16kHz target.
 */
export function linearResample(samples: Float32Array, fromRate: number, toRate: number): Float32Array {
  if (fromRate === toRate) return samples;
  const ratio = fromRate / toRate;
  const newLength = Math.round(samples.length / ratio);
  const result = new Float32Array(newLength);
  for (let i = 0; i < newLength; i++) {
    const srcIndex = i * ratio;
    const lower = Math.floor(srcIndex);
    const upper = Math.min(lower + 1, samples.length - 1);
    const frac = srcIndex - lower;
    result[i] = samples[lower] * (1 - frac) + samples[upper] * frac;
  }
  return result;
}

/**
 * Convert Float32 audio samples (-1.0 to 1.0) to 16-bit PCM Int16Array.
 */
export function float32ToPcm16(float32: Float32Array): Int16Array {
  const pcm16 = new Int16Array(float32.length);
  for (let i = 0; i < float32.length; i++) {
    const clamped = Math.max(-1, Math.min(1, float32[i]));
    pcm16[i] = clamped < 0 ? clamped * 0x8000 : clamped * 0x7fff;
  }
  return pcm16;
}

/**
 * Convert ArrayBuffer to base64 string for sending over WebSocket.
 */
export function arrayBufferToBase64(buffer: ArrayBuffer): string {
  const bytes = new Uint8Array(buffer);
  let binary = "";
  for (let i = 0; i < bytes.byteLength; i++) {
    binary += String.fromCharCode(bytes[i]);
  }
  return btoa(binary);
}
```

```typescript
import { GoogleGenAI, Modality } from "@google/genai";
import type { Session, LiveServerMessage } from "@google/genai";

export interface GeminiLiveCallbacks {
  onOpen: () => void;
  onText: (text: string) => void;
  onTurnComplete: () => void;
  onError: (error: Event) => void;
  onClose: () => void;
}

/**
 * Connect to Gemini Live API using an ephemeral token.
 * Returns the Session for sending audio and closing.
 */
export async function connectToGeminiLive(
  ephemeralToken: string,
  callbacks: GeminiLiveCallbacks,
): Promise<Session> {
  const ai = new GoogleGenAI({
    apiKey: ephemeralToken,
    httpOptions: { apiVersion: "v1alpha" },
  });

  const session = await ai.live.connect({
    model: "gemini-2.0-flash",
    config: { responseModalities: [Modality.TEXT] },
    callbacks: {
      onopen: () => callbacks.onOpen(),
      onmessage: (msg: LiveServerMessage) => {
        if (msg.text) {
          callbacks.onText(msg.text);
        }
        if (msg.serverContent?.turnComplete) {
          callbacks.onTurnComplete();
        }
      },
      onerror: (e: Event) => callbacks.onError(e),
      onclose: () => callbacks.onClose(),
    },
  });

  return session;
}
```

Implementation details:
- `start()`:
  1. Set status to "connecting", clear analysis and error
  2. Fetch `POST /api/token` with `{ referenceText }`. On failure → set error to "Unable to connect to AI coach. Check your internet connection and try again." (per D-20), set status "error", return.
  3. Call `connectToGeminiLive(token, callbacks)` where:
     - `onOpen`: set status "recording"
     - `onText`: append text to analysis state (per D-06 incremental rendering)
     - `onTurnComplete`: set status "complete"
     - `onError`: set error "Connection lost — please try again" (per D-11), set status "error"
     - `onClose`: if status is still "recording", set to "complete"
  4. Store session in a ref
  5. Call `startAudioCapture(session)` — internal function that:
     - Calls `navigator.mediaDevices.getUserMedia({ audio: true })`. On NotAllowedError → set error to "Microphone access is required. Please allow microphone access in your browser settings." (per D-21), set status "error", close session, return.
     - Creates AudioContext, MediaStreamSource, loads worklet from `/audio-worklet-processor.js`
     - Creates AudioWorkletNode("pcm-processor"), connects source → workletNode

  6. Store audioCtx and stream refs for cleanup

- `stop()`:
  1. Close session ref (`session.close()`)
  2. Stop all tracks on the media stream (`stream.getTracks().forEach(t => t.stop())`)
  3. Close AudioContext
  4. If status was "recording", set to "complete"

- Cleanup on unmount: call stop() if session is active
